cmake_minimum_required(VERSION 4.0)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
    project(LumaRecodeData VERSION 0.1.0 LANGUAGES C)
    set(CMAKE_C_STANDARD 17)
    set(CMAKE_C_STANDARD_REQUIRED ON)
    set(CMAKE_C_EXTENSIONS OFF)
    include(CTest)
    set(luma_c_standalone ON)
endif()

add_library(luma_c OBJECT data.c)
set_target_properties(luma_c PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(luma_c PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
if(luma_c_standalone)
    target_compile_definitions(luma_c PRIVATE LUMA_CORE_STATIC)
    add_library(luma_core_data STATIC $<TARGET_OBJECTS:luma_c>)
    target_include_directories(luma_core_data PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
    target_compile_definitions(luma_core_data INTERFACE LUMA_CORE_STATIC)
    if(BUILD_TESTING)
        add_executable(luma_c_checks ../tests/c_checks.c)
        target_link_libraries(luma_c_checks PRIVATE luma_core_data)
        add_test(NAME c_contracts COMMAND luma_c_checks)
    endif()
else()
    target_compile_definitions(luma_c PRIVATE LUMA_CORE_BUILD)
endif()
